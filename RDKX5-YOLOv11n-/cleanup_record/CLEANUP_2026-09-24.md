```bash
# 1) PTQ 校准集（~1 分钟）

    --coco-path data/derived/AUV_5_selected_3000_Dwb --output-dir calibration_data --num-images 300

# 2) 域对比数据集（6 个，各 1302 张，逐帧同划分）

    --prov runs/prov/provenance_final_PNP.kpt4.yolov8.csv
for D in B C D; do for E in on off; do
  python scripts/1_prepare/pose/map_pose_dataset.py --domain $D --enhance $E \
      --out experiment/data/pose_${D}$([ $E = off ] && echo _noenh)
done; done
#   （定稿生产集另建： --domain D --enhance wb --out data/mapped/pose_D_wb）

# 3) 域/消融训练可视化：重跑对应脚本即可
bash experiment/scripts/train_domain_arms.sh                 # 四臂（产物 weights/domain_*.pt）
python experiment/scripts/exp_aug/exp_aug_ablation.py        # 增强消融两臂（产物 *_best.pt）
# 4) 标定 QC 图：重跑标定流程（见 scripts/1_prepare/calibrate_camera.py --save-views）
```
